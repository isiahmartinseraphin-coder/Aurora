local function round(object, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius)
    c.Parent = object
end

local function border(object, color, thickness)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness or 1
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = object
end

local function makeText(parent, value, size, position)
    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Text = value
    label.TextColor3 = COLORS.white
    label.TextSize = size
    label.Font = Enum.Font.Code
    label.Position = position
    label.Size = UDim2.new(1, 0, 0, size + 8)
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.TextYAlignment = Enum.TextYAlignment.Center
    label.Parent = parent

    return label
end

--==================================================
-- DRAG SYSTEM
--==================================================

local function draggable(frame, handle)

    local dragging = false
    local dragStart
    local startPosition

    handle.InputBegan:Connect(function(input)

        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

            dragging = true
            dragStart = input.Position
            startPosition = frame.Position

            input.Changed:Connect(function()

                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end

            end)
        end
    end)

    UIS.InputChanged:Connect(function(input)

        if not dragging then
            return
        end

        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then

            local delta = input.Position - dragStart

            frame.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end)
end

local function createWindow(name, position, size)

    local frame = Instance.new("Frame")
    frame.Name = name
    frame.Position = position
    frame.Size = size
    frame.BackgroundColor3 = COLORS.panel
    frame.BackgroundTransparency = 0.04
    frame.BorderSizePixel = 0
    frame.Parent = gui

    round(frame, 9)
    border(frame, COLORS.purple, 1.8)

    -- Glow
    local glow = Instance.new("UIStroke")
    glow.Color = COLORS.purple
    glow.Thickness = 5
    glow.Transparency = 0.88
    glow.Parent = frame

    -- Header
    local header = Instance.new("Frame")
    header.Size = UDim2.new(1, 0, 0, 35)
    header.BackgroundColor3 = COLORS.header
    header.BorderSizePixel = 0
    header.Parent = frame

    round(header, 8)

    local separator = Instance.new("Frame")
    separator.Position = UDim2.fromOffset(7, 34)
    separator.Size = UDim2.new(1, -14, 0, 1)
    separator.BackgroundColor3 = COLORS.purpleDark
    separator.BorderSizePixel = 0
    separator.Parent = header

    local title = makeText(
        header,
        name,
        16,
        UDim2.fromOffset(11, 1)
    )

    title.TextColor3 = COLORS.purple

    -- Minimize
    local minimize = Instance.new("TextButton")
    minimize.BackgroundTransparency = 1
    minimize.Text = "−"
    minimize.TextColor3 = COLORS.gray
    minimize.TextSize = 19
    minimize.Font = Enum.Font.Code
    minimize.Position = UDim2.new(1, -57, 0, 2)
    minimize.Size = UDim2.fromOffset(25, 27)
    minimize.AutoButtonColor = false
    minimize.Parent = header

    -- Close
    local close = Instance.new("TextButton")
    close.BackgroundTransparency = 1
    close.Text = "×"
    close.TextColor3 = COLORS.gray
    close.TextSize = 20
    close.Font = Enum.Font.Code
    close.Position = UDim2.new(1, -30, 0, 1)
    close.Size = UDim2.fromOffset(25, 27)
    close.AutoButtonColor = false
    close.Parent = header

    -- Content
    local content = Instance.new("Frame")
    content.BackgroundTransparency = 1
    content.Position = UDim2.fromOffset(8, 42)
    content.Size = UDim2.new(1, -16, 1, -49)
    content.Parent = frame

    local originalSize = size
    local minimized = false

    minimize.MouseButton1Click:Connect(function()

        minimized = not minimized

        if minimized then

            content.Visible = false

            TweenService:Create(
                frame,
                TweenInfo.new(.18),
                {
                    Size = UDim2.new(
                        originalSize.X.Scale,
                        originalSize.X.Offset,
                        0,
                        35
                    )
                }
            ):Play()

        else

            content.Visible = true

            TweenService:Create(
                frame,
                TweenInfo.new(.18),
                {
                    Size = originalSize
                }
            ):Play()
        end
    end)

    close.MouseButton1Click:Connect(function()

        TweenService:Create(
            frame,
            TweenInfo.new(.15),
            {
                Size = UDim2.fromOffset(0, 0)
            }
        ):Play()

        task.delay(.16, function()
            if frame.Parent then
                frame:Destroy()
            end
        end)
    end)

    draggable(frame, header)

    return frame, content
end

local function createButton(parent, value, y, height)

    local button = Instance.new("TextButton")
    button.Position = UDim2.fromOffset(0, y)
    button.Size = UDim2.new(1, 0, 0, height or 38)
    button.BackgroundColor3 = COLORS.button
    button.Text = value
    button.TextColor3 = COLORS.white
    button.TextSize = 16
    button.Font = Enum.Font.Code
    button.AutoButtonColor = false
    button.Parent = parent

    round(button, 8)
    border(button, COLORS.blue, 1.25)

    button.MouseEnter:Connect(function()

        TweenService:Create(
            button,
            TweenInfo.new(.1),
            {
                BackgroundColor3 = Color3.fromRGB(37, 30, 50)
            }
        ):Play()
    end)

    button.MouseLeave:Connect(function()

        TweenService:Create(
            button,
            TweenInfo.new(.1),
            {
                BackgroundColor3 = COLORS.button
            }
        ):Play()
    end)

    return button
end

local function createToggle(parent, name, y)

    local holder = Instance.new("Frame")
    holder.Position = UDim2.fromOffset(0, y)
    holder.Size = UDim2.new(1, 0, 0, 39)
    holder.BackgroundColor3 = COLORS.button
    holder.Parent = parent

    round(holder, 7)
    border(holder, COLORS.blue, 1.2)

    makeText(
        holder,
        name,
        15,
        UDim2.fromOffset(11, 1)
    )

    local toggle = Instance.new("TextButton")
    toggle.Position = UDim2.new(1, -69, 0, 7)
    toggle.Size = UDim2.fromOffset(55, 25)
    toggle.BackgroundColor3 = Color3.fromRGB(48, 48, 54)
    toggle.Text = ""
    toggle.AutoButtonColor = false
    toggle.Parent = holder

    round(toggle, 20)

    local knob = Instance.new("Frame")
    knob.Position = UDim2.fromOffset(3, 3)
    knob.Size = UDim2.fromOffset(19, 19)
    knob.BackgroundColor3 = COLORS.white
    knob.Parent = toggle

    round(knob, 20)

    local enabled = false

    toggle.MouseButton1Click:Connect(function()

        enabled = not enabled

        TweenService:Create(
            toggle,
            TweenInfo.new(.15),
            {
                BackgroundColor3 =
                    enabled
                    and COLORS.purple
                    or Color3.fromRGB(48, 48, 54)
            }
        ):Play()

        TweenService:Create(
            knob,
            TweenInfo.new(.15),
            {
                Position =
                    enabled
                    and UDim2.fromOffset(33, 3)
                    or UDim2.fromOffset(3, 3)
            }
        ):Play()
    end)

    return holder
end
